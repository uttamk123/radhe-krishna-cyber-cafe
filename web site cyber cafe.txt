      function getDocumentFilesError(files) {
        if (files.length === 0) return "????? ??-??-?? ?? ????????? ??????";
        if (files.length > maxDocumentCount) return `?????? ${maxDocumentCount} ??????? ??????`;

        let totalSize = 0;
        for (const file of files) {
          const extension = file.name.split(".").pop().toLowerCase();
          if (!allowedDocumentExtensions.has(extension)) {
            return "???? PDF, JPG, PNG, DOC ?? DOCX ??????? ??? ???? ????";
          }
          if (file.size > maxDocumentSize) {
            return `${file.name}: ????? 10 MB ?? ???? ???`;
          }
          totalSize += file.size;
        }

        if (totalSize > maxTotalDocumentSize) return "???? ?? ??????? ?? ??? ???? 20 MB ?? ?? ?????";
        return "";
      }
